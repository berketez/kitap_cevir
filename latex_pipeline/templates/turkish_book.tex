\newXeTeXintercharclass\kc@fallbackclass
\def\kc@fallbacktoks#1{%
  \XeTeXinterchartoks#1\kc@fallbackclass=\expandafter{%
    \the\XeTeXinterchartoks#1\z@\begingroup\kc@fallbackfamily}%
  \XeTeXinterchartoks\kc@fallbackclass#1=\expandafter{%
    \expandafter\endgroup\the\XeTeXinterchartoks\z@#1}}
\def\kc@fallbackscan#1#2{%
  \kc@cp=#1\relax
  \loop
    \iffontchar\font\kc@cp\else
      \iffontchar\kc@fallbackprobe\kc@cp
        \global\XeTeXcharclass\kc@cp=\kc@fallbackclass
      \fi
    \fi
  \ifnum\kc@cp<#2\relax \advance\kc@cp\@ne \repeat}
% polyglossia dili (ve Türkçe noktalama geçişlerini) begindocument/end'de

      \kc@fallbackscan{"1F00}{"1FFF}% Yunanca (çok tonlu)
      \kc@fallbackscan{"2010}{"205E}% genel noktalama (′ ″ ‰)
      \kc@fallbackscan{"2070}{"20CF}% üst/alt simge, para birimi
      \kc@fallbackscan{"2100}{"23FF}% harf benzeri, sayı biçimi, ok, matematik
      \kc@fallbackscan{"25A0}{"25FF}% geometrik şekil (◦)
      \kc@fallbackscan{"2A00}{"2AFF}% ek matematik işlemleri
      \kc@fallbackscan{"1D400}{"1D7FF}% matematik harf ve rakamları
    \endgroup
    % Her sınıftan (4095 = sınır) yedek sınıfa geçişte grup açılır, çıkışta

      \kc@fallbacktoks\kc@cp
    \ifnum\kc@cp<\numexpr\kc@fallbackclass-\@ne\relax \advance\kc@cp\@ne \repeat

    \XeTeXinterchartokenstate=\@ne
  \fi
}

% #1: kaynak sınıf, #2: yedek sınıf, #3: yedek aile (geçişler birinci
% yedekteki gibi; sınıf 0'ın geçiş dizgisi grubun dışında korunur).
\def\kc@classtoks#1#2#3{%
  \XeTeXinterchartoks#1#2=\expandafter{%
    \the\XeTeXinterchartoks#1\z@\begingroup#3}%
  \XeTeXinterchartoks#2#1=\expandafter{%
    \expandafter\endgroup\the\XeTeXinterchartoks\z@#1}}
% #1–#2: harf aralığı, #3: yoklama yazı tipi, #4: sınıf. Ana yazı tipinde
% olmayan ve henüz bir yedeğe atanmamış (sınıf 0) harfler.
\def\kc@classscan#1#2#3#4{%
  \kc@cp=#1\relax
  \loop
    \iffontchar\font\kc@cp\else
      \ifnum\XeTeXcharclass\kc@cp=\z@
        \iffontchar#3\kc@cp
          \global\XeTeXcharclass\kc@cp=#4\relax
        \fi
      \fi
    \fi
  \ifnum\kc@cp<#2\relax \advance\kc@cp\@ne \repeat}
% #1: sınıf, #2: aile — her sınıftan (ve 4095 sınırından) geçiş.
\def\kc@classloop#1#2{%
  \kc@cp=\z@
  \loop
    \kc@classtoks\kc@cp#1#2%
  \ifnum\kc@cp<\numexpr#1-\@ne\relax \advance\kc@cp\@ne \repeat
  \kc@classtoks{4095}#1#2}
\AddToHook{begindocument/end}[kitapcevir/simge]{%
  \begingroup
    \normalfont
    \ifnum\kc@haveemoji=\@ne
      \kc@classscan{"2600}{"27BF}\kc@emojiprobe\kc@emojiclass% simge, dingbat
      \kc@classscan{"2B00}{"2BFF}\kc@emojiprobe\kc@emojiclass% simge ve oklar
      \kc@classscan{"1F300}{"1FAFF}\kc@emojiprobe\kc@emojiclass% emoji
    \fi
    \ifnum\kc@havesymbol=\@ne
      \kc@classscan{"2600}{"27BF}\kc@symbolprobe\kc@symbolclass
      \kc@classscan{"2B00}{"2BFF}\kc@symbolprobe\kc@symbolclass
    \fi
  \endgroup
  \ifnum\kc@haveemoji=\@ne \kc@classloop\kc@emojiclass\kc@emojifamily \fi
  \ifnum\kc@havesymbol=\@ne \kc@classloop\kc@symbolclass\kc@symbolfamily \fi
  \XeTeXinterchartokenstate=\@ne
}

% Kaynak PDF'te çerçeve ya da renkli zemin içindeki yan metin ("Exkurs",
% yasalar kutusu) dönüştürücüde \begin{kckutu} … \end{kckutu} olur
% (latex_pipeline/kutu_onarim.py). Yalnız çekirdek LaTeX ve zaten yüklü
% xcolor kullanılır: paketlenen hafif TeX'in (tectonic) ısıtma önbelleğine
% yeni dosya gerekmez. İçerik önce kutu genişliğinde bir \vbox'ta dizilir.
% Yüksekliği \kckutusiniri'ni (metin alanının yarısı) aşmıyorsa gri zeminli,
% çerçeveli tek parça basılır (sayfaya sığmazsa sonraki sayfaya geçer);
% aşıyorsa bölünemeyen kutu sayfadan taşacağı için içerik \unvbox ile akışa
% döner (sayfalar arasında bölünür) ve yalnız üst/alt çizgiyle ayrılır:
% metin hiçbir durumda kesilmez. Kutu başındaki başlığın üst boşluğu
% minipage'deki gibi yutulur (\@minipagetrue). Kayan nesne, tablo, dipnot
% ve bölüm başlığı içeren kutuyu dönüştürücü zaten basmaz.
\definecolor{kckutucerceve}{gray}{0.45}
\definecolor{kckutuzemin}{gray}{0.95}
\newsavebox{\kckutubox}
\newlength{\kckutusiniri}
\makeatletter
\newenvironment{kckutu}{%
  \par\addvspace{\medskipamount}%
  \setlength{\fboxsep}{6pt}\setlength{\fboxrule}{0.6pt}%
  \setlength{\kckutusiniri}{0.5\textheight}%
  \setbox\kckutubox\vbox\bgroup
    \hsize\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax
    \@parboxrestore
    \@minipagetrue\everypar{\@minipagefalse\everypar{}}%
    \setlength{\parskip}{0.5\baselineskip}%
}{%
  \par\egroup
  \ifdim\dimexpr\ht\kckutubox+\dp\kckutubox\relax>\kckutusiniri
    \noindent{\color{kckutucerceve}\rule{\linewidth}{\fboxrule}}\par
    \nobreak\vskip\fboxsep
    \unvbox\kckutubox\par
    \vskip\fboxsep
    \noindent{\color{kckutucerceve}\rule{\linewidth}{\fboxrule}}\par
  \else
    \noindent\fcolorbox{kckutucerceve}{kckutuzemin}{\usebox{\kckutubox}}\par
  \fi
  \addvspace{\medskipamount}%
}
\makeatother
